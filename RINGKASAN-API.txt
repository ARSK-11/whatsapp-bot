1️⃣  KIRIM PESAN SEDERHANA:

    curl -X POST http://localhost:3000/send \
      -H "Content-Type: application/json" \
      -d '{"number":"089509844606","message":"Halo!"}'


2️⃣  KIRIM PESAN DENGAN EMOJI:

    curl -X POST http://localhost:3000/send \
      -H "Content-Type: application/json" \
      -d '{"number":"089509844606","message":"👋 Hi! ✅"}'


3️⃣  KIRIM GAMBAR:

    curl -X POST http://localhost:3000/send-media \
      -H "Content-Type: application/json" \
      -d '{"number":"089509844606","media_url":"https://picsum.photos/500","caption":"Gambar test"}'


4️⃣  BROADCAST:

    curl -X POST http://localhost:3000/broadcast \
      -H "Content-Type: application/json" \
      -d '{"numbers":["089509844606","081234567890"],"message":"Broadcast!"}'


5️⃣  CEK STATUS:

    curl http://localhost:3000/status

┌─────────────────────────────────────────────────────────────────┐
│ PHP                                                              │
└─────────────────────────────────────────────────────────────────┘

<?php
$ch = curl_init('http://localhost:3000/send');
curl_setopt($ch, CURLOPT_POST, 1);
curl_setopt($ch, CURLOPT_POSTFIELDS, json_encode([
    'number' => '089509844606',
    'message' => 'Halo dari PHP!'
]));
curl_setopt($ch, CURLOPT_HTTPHEADER, ['Content-Type: application/json']);
curl_exec($ch);
?>
